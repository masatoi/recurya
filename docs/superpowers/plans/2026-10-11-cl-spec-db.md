# cl-spec DB Layer Contracts Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Contract the DB layer with cl-spec — Function Specs for the pure helpers in `db/core` / `db/users`, and test-DB-isolated Properties for course-notebook ordering, learn progress/code storage, and notebook/course CRUD — and fix the bugs the contracts expose (F-C1 move across a position gap, F-C3 non-ASCII digits in `uuid-string-p`, F-C2 docstring).

**Architecture:** New contract files under `specs/db/` (packages `recurya/specs/db/*`, never loaded by production). Every DB-touching contract is a `defproperty` whose body wraps each trial in `recurya/tests/support/db:with-test-db`, so it only ever touches `recurya_test` whichever runner (Rove harness or the `spec-check` tool, which runs on its own thread) executes it. Generators return plans (operation lists, value plists); rows are created inside the body. The existing harness `tests/specs/contracts.lisp` runs them once their names are in its expected lists.

**Tech Stack:** SBCL, cl-spec (`cl-spec/main`, check-it backend), Rove, Mito, cl-dbi, PostgreSQL (`recurya_test`), local-time, cl-ppcre, package-inferred ASDF, cl-mcp tools.

Design: `docs/superpowers/specs/2026-10-11-cl-spec-db-design.md`.

;;;;
;;;; Column value domains taken from db/schema.sql and the route-level allow
;;;; lists, generators for in-domain values, and small helpers shared by the
;;;; DB contract files.  Not loaded by the production system.

(defpackage #:recurya/specs/db/domain
  (:use #:cl)
  (:import-from #:cl-spec/main
                #:defspec
                #:defgenerator)
  (:import-from #:cl-spec/src/backends/check-it)
  (:import-from #:uuid

  (:export #:*publication-statuses*

           #:publication-status
           #:publication-status-p
           #:visibility
           #:visibility-p
           #:varchar-255
           #:varchar-255-p
           #:summary-text
           #:summary-text-p

           #:random-db-text
           #:random-summary
           #:id-string
           #:fresh-user-id
           #:set-equal-p))

(defparameter *publication-statuses* '("draft" "published")
  "Notebook and course statuses: the STATUS values that %DECODE-STATE-TOKEN in
web/routes.lisp produces.")

(defparameter *visibilities* '("private" "unlisted" "public")
  "Notebook and course visibilities: %DECODE-STATE-TOKEN's values and the allow
list in NOTEBOOK-UPDATE-HANDLER (web/routes.lisp).")

(defparameter *text-pieces*
  (vector "a" "Title" "ラムダ λ" "say \"hi\"" "back\\slash" "x-y_z")
  "Pieces of in-domain text: ASCII, non-ASCII, quotes, backslash, punctuation.")

(defun random-element (sequence)
  "Return a uniformly chosen element of the non-empty SEQUENCE."
  (elt sequence (random (length sequence))))

(defun random-db-text (max-length)
  "A non-empty string of at most MAX-LENGTH characters; a fifth of the time
exactly MAX-LENGTH characters (ASCII or non-ASCII), to probe the column limit."
  (if (zerop (random 5))
      (make-string max-length :initial-element (random-element #(#\a #\λ)))
      (let ((text (format nil "~{~A~^ ~}"
                          (loop repeat (1+ (random 3))
                                collect (random-element *text-pieces*)))))
        (subseq text 0 (min max-length (length text))))))

(defun random-summary ()
  "NIL, the empty string, or text of at most 500 characters."
  (case (random 4)
    (0 nil)
    (1 "")
    (t (random-db-text 500))))

(defun publication-status-p (value)
  "True when VALUE is one of *PUBLICATION-STATUSES*."
  (and (stringp value) (member value *publication-statuses* :test #'string=) t))

(defun visibility-p (value)
  "True when VALUE is one of *VISIBILITIES*."
  (and (stringp value) (member value *visibilities* :test #'string=) t))

(defun varchar-255-p (value)
  "True for a string of 1-255 characters: the VARCHAR(255) NOT NULL columns
(title, slug) of db/schema.sql."
  (and (stringp value) (<= 1 (length value) 255)))

(defun summary-text-p (value)
  "True for NIL or a string of at most 500 characters: summary VARCHAR(500) NULL."
  (or (null value)
      (and (stringp value) (<= (length value) 500))))

(defgenerator publication-status ()
  "One of *PUBLICATION-STATUSES*."
  (random-element *publication-statuses*))

(defspec publication-status (satisfies publication-status-p)
  (:generator publication-status))

(defgenerator visibility ()
  "One of *VISIBILITIES*."
  (random-element *visibilities*))

(defspec visibility (satisfies visibility-p)
  (:generator visibility))

(defgenerator varchar-255 ()
  "In-domain title or slug text."
  (random-db-text 255))

(defspec varchar-255 (satisfies varchar-255-p)
  (:generator varchar-255))

(defgenerator summary-text ()
  "In-domain summary: NIL, empty, or up to 500 characters."
  (random-summary))

(defspec summary-text (satisfies summary-text-p)
  (:generator summary-text))

(defun id-string (id)
  "ID (a UUID object or string) as a lower-case string, for comparing ids."
  (string-downcase (princ-to-string id)))

(defun fresh-user-id ()
  "A fresh lower-case UUID string.  The learn tables key users by UUID without a
foreign key, so no user row is needed."
  (id-string (make-v4-uuid)))

(defun set-equal-p (a b)
  "True when lists A and B hold the same elements under EQUAL, ignoring order
and multiplicity."
  (and (subsetp a b :test #'equal)
       (subsetp b a :test #'equal)))

  (:import-from #:cl-ppcre
                #:scan)
  (:import-from #:uuid
                #:make-v4-uuid)
  (:import-from #:local-time
                #:decode-timestamp
                #:unix-to-timestamp))
(in-package #:recurya/specs/db/core)

(defgenerator db-value ()
  "A value that may reach or leave cl-dbi: NIL, :NULL, numbers, strings,
keywords, plain symbols, a float, a list and a character."
  (random-element (vector nil :null 0 42 "" "x" :pending 'plain-symbol 1.5
                          (list 1 2) #\a)))

(defspec db-value t (:generator db-value))

(defspec-function nil->null
  "NIL becomes :NULL; every other value is returned unchanged (EQ)."
  (:args (value db-value))
  (:post (if (null value) (eq result :null) (eq result value))))

(defspec-function null->nil
  ":NULL becomes NIL; every other value is returned unchanged (EQ)."
  (:args (value db-value))
  (:post (if (eq value :null) (null result) (eq result value))))

(defspec-function keyword->db-string
  "NIL gives NIL, a keyword gives its name in lower case, and any other value
is refused with a TYPE-ERROR (the docstring's pre-condition)."
  (:args (kw db-value))
  (:cases
    (:absent
      (:when (null kw))
      (:returns null))
    (:keyword
      (:when (keywordp kw))
      (:returns string)
      (:post (string= result (string-downcase (symbol-name kw)))))
    (:other
      (:when (and kw (not (keywordp kw))))
      (:signals (type type-error)))))

(defgenerator db-timestamp ()
  "A timestamp between 1970 and 2096 with a random microsecond."
  (unix-to-timestamp (random 4000000000) :nsec (* 1000 (random 1000000))))

(defspec db-timestamp (type local-time:timestamp) (:generator db-timestamp))

(defun expected-db-timestamp (ts)
  "TS as the docstring's 'YYYY-MM-DD HH:MM:SS.UUUUUU' in the default time zone,
built from DECODE-TIMESTAMP's fields rather than a format string."
  (multiple-value-bind (nsec sec min hour day month year) (decode-timestamp ts)
    (format nil "~4,'0D-~2,'0D-~2,'0D ~2,'0D:~2,'0D:~2,'0D.~6,'0D"
            year month day hour min sec (floor nsec 1000))))

(defspec-function format-timestamp-for-db
  "A timestamp is written as 'YYYY-MM-DD HH:MM:SS.UUUUUU' with microseconds."
  (:args (ts db-timestamp))
  (:returns string)
  (:post (string= result (expected-db-timestamp ts))))

(defun random-uuid-string ()
  "A fresh lower-case UUID string (of CHARACTERs, so it can be mutated)."
  (map 'string #'char-downcase (princ-to-string (make-v4-uuid))))

(defgenerator uuid-like-input ()
  "A UUID object, a non-string value, or a UUID string in either case padded
with spaces."
  (case (random 4)
    (0 (make-v4-uuid))
    (1 (random-element (vector 42 :abc 'plain-symbol)))
    (t (let ((text (random-uuid-string)))
         (format nil "~A~A~A"
                 (random-element #("" " " "  "))
                 (if (zerop (random 2)) (string-upcase text) text)
                 (random-element #("" " " "   ")))))))

(defspec uuid-like-input t (:generator uuid-like-input))

(defspec-function ensure-uuid
  "A string is trimmed of outer spaces and lower-cased; any other value is
printed and lower-cased."
  (:args (value uuid-like-input))
  (:returns string)
  (:post (string= result
                  (string-downcase (if (stringp value)
                                       (string-trim " " value)
                                       (princ-to-string value))))))

(defparameter *uuid-shape*
  (concatenate 'string
               "\\A[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}"
               "-[0-9a-fA-F]{12}\\z")
  "The 8-4-4-4-12 hexadecimal shape PostgreSQL accepts for a UUID: ASCII hex
digits of either case only.")

(defun mutate-uuid (text)
  "TEXT, a UUID string, with one defect (or none): a non-hex or non-ASCII digit
character, upper case, a missing or extra character, a trailing newline, no
hyphens, or a leading space."
  (let ((copy (copy-seq text))
        (i (random (length text))))
    (case (random 8)
      (0 (setf (char copy i)
               (random-element (vector #\g #\z #\- #\Space #\G
                                       (code-char #x0661) (code-char #xFF11))))
         copy)
      (1 (string-upcase copy))
      (2 (subseq copy 1))
      (3 (concatenate 'string copy "0"))
      (4 (concatenate 'string copy (string #\Newline)))
      (5 (remove #\- copy))
      (6 (concatenate 'string " " copy))
      (t copy))))

(defgenerator uuid-candidate ()
  "A value that may or may not be a canonical UUID string."
  (case (random 5)
    (0 (random-element (vector nil 42 :uuid "" "nil")))
    (1 (random-uuid-string))
    (t (mutate-uuid (random-uuid-string)))))

(defspec uuid-candidate t (:generator uuid-candidate))

(defspec-function uuid-string-p
  "True exactly for a 36-character 8-4-4-4-12 string of ASCII hex digits, so a
value PostgreSQL would reject as a UUID never reaches it."
  (:args (value uuid-candidate))
  (:post (eq (and result t)
             (and (stringp value) (scan *uuid-shape* value) t))))

(defparameter *placeholder-shape* "\\Au-[0-9a-f]{8}\\z"
  "The placeholder handle shape from PLACEHOLDER-HANDLE-P's docstring: u- and
exactly 8 lower-case hex digits.")

(defgenerator handle-candidate ()
  "A value that may or may not be a placeholder handle."
  (let ((hex (subseq (random-uuid-string) 0 8)))
    (case (random 7)
      (0 (format nil "u-~A" hex))
      (1 (format nil "u-~A" (string-upcase hex)))
      (2 (format nil "u-~A" (subseq hex 1)))
      (3 (format nil "u-~A0" hex))
      (4 (format nil "U-~A" hex))
      (5 (format nil "u-~A~%" hex))
      (t (random-element (vector nil "" "alice" "u-" 42
                                 (format nil "u-~Ag" (subseq hex 1))))))))

(defspec handle-candidate t (:generator handle-candidate))

(defspec-function placeholder-handle-p
  "True exactly for u- followed by 8 lower-case hex digits."
  (:args (handle handle-candidate))
  (:post (eq (and result t)
             (and (stringp handle) (scan *placeholder-shape* handle) t))))

(defproperty generated-placeholder-handle-is-placeholder ()
  "Every handle %GENERATE-PLACEHOLDER-HANDLE makes has the placeholder shape, so
a new OAuth user is always routed to onboarding."
  (:about placeholder-handle-p)
  (:kind :invariant)
  (:trials (:smoke 50 :normal 200))
  (let ((handle (%generate-placeholder-handle)))
    (and (placeholder-handle-p handle)
         (scan *placeholder-shape* handle)
         t)))
```

(defun uuid-string-p (value)
  "Return T if VALUE is a canonical 8-4-4-4-12 hexadecimal UUID string.

Used to guard DB id lookups so a malformed id (e.g. the literal \"nil\" a
broken form action can send to a /.../:id route) resolves to \"not found\"
instead of reaching PostgreSQL and signalling an \"invalid input syntax for
type uuid\" error (SQLSTATE 22P02) that would otherwise crash the request.
Only ASCII hex digits count: DIGIT-CHAR-P would also accept other Unicode
digits (e.g. Arabic-Indic or fullwidth), which PostgreSQL rejects."
  (and (stringp value)
       (= (length value) 36)
       (loop for ch across value
             for i from 0
             always (if (member i '(8 13 18 23))
                        (char= ch #\-)
                        (find ch "0123456789abcdefABCDEF")))))
```

(defgenerator course-notebook-ops ()
  "8-20 operations (KIND N): KIND is :ADD (twice as likely), :REMOVE, :UP or
:DOWN, and N picks the target among the candidates modulo their count."
  (loop repeat (+ 8 (random 13))
        collect (list (random-element #(:add :add :remove :up :down))
                      (random *notebook-count*))))

(defspec course-notebook-ops list (:generator course-notebook-ops))

(defun row-notebook-id (row)

  "The notebook ids of COURSE-ID's join rows in listed (position) order."
  (mapcar #'row-notebook-id (list-course-notebooks course-id)))

(defun row-id-of (course-id notebook-id)
  "The join-row id linking COURSE-ID to NOTEBOOK-ID."

   (find notebook-id (list-course-notebooks course-id)
         :key #'row-notebook-id :test #'string=)))

the one the docstrings promise (a row for :ADD; T when a row was removed or
moved; NIL for moving the first row up or the last row down)."
  (destructuring-bind (kind n) op
    (if (eq kind :add)
        (let ((free (remove-if (lambda (id) (member id model :test #'string=))
                               notebook-ids)))
          (if (null free)
              (values t model)
              (let ((id (nth (mod n (length free)) free)))

                        (append model (list id))))))

                   (id (nth k model))

                (:remove

            (codes (make-hash-table :test 'equal))

                                                      (gethash (cons n c) codes)))))))))))

(defgenerator listing-scenario ()
  "(ROWS QUERY OFFSET LIMIT): 0-8 rows (AUTHOR STATUS VISIBILITY) with AUTHOR an
index into two authors; a QUERY plist of optional :STATUS, :AUTHOR (index) and
:VISIBILITY; an OFFSET of NIL, 0, 1 or 3; and a LIMIT of :DEFAULT (argument
omitted), NIL, 0, 1, 2 or 5."
  (list (loop repeat (random 9)
              collect (list (random 2) (random-status) (random-visibility)))
        (list :status (random-element (vector nil "draft" "published"))
              :author (random-element #(nil 0 1))
              :visibility (random-element (vector nil "private" "unlisted" "public")))
        (random-element #(nil 0 1 3))
        (random-element #(:default nil 0 1 2 5))))

(defspec listing-scenario list (:generator listing-scenario))

(defun listing-holds-p (scenario create list-fn count-fn id-fn created-at-fn)
  "Create SCENARIO's rows with CREATE (AUTHOR SLUG STATUS VISIBILITY -> row) and
check LIST-FN and COUNT-FN against filtering the created rows in Lisp: with no
limit the listing is exactly the matching rows, newest first; the count is
their number; and an OFFSET/LIMIT listing is the slice [OFFSET, OFFSET+LIMIT)
of it (LIMIT defaulting to 50)."
  (destructuring-bind (rows query offset limit) scenario
    (let* ((authors (vector (create-test-user) (create-test-user)))
           (author-ids (map 'vector (lambda (author) (id-string (users-id author)))
                            authors))
           (created (loop for (a status visibility) in rows
                          for i from 0
                          collect (list (id-string
                                         (funcall id-fn
                                                  (funcall create (aref authors a)
                                                           (format nil "slug-~D" i)
                                                           status visibility)))
                                        a status visibility)))
           (q-status (getf query :status))
           (q-author (getf query :author))
           (q-visibility (getf query :visibility))
           (filters (append (and q-status (list :status q-status))
                            (and q-author (list :author-id (aref author-ids q-author)))
                            (and q-visibility (list :visibility q-visibility))))
           (want (loop for (id a status visibility) in created
                       when (and (or (null q-status) (string= status q-status))
                                 (or (null q-author) (= a q-author))
                                 (or (null q-visibility) (string= visibility q-visibility)))
                         collect id))
           (full (apply list-fn :limit nil filters))
           (full-ids (mapcar (lambda (row) (id-string (funcall id-fn row))) full))
           (paged (apply list-fn (append (and offset (list :offset offset))
                                         (unless (eq limit :default) (list :limit limit))
                                         filters)))
           (start (min (or offset 0) (length full)))
           (end (let ((size (if (eq limit :default) 50 limit)))
                  (if size (min (+ start size) (length full)) (length full)))))
      (and (= (length full-ids) (length want))
           (set-equal-p full-ids want)
           (loop for (a b) on full
                 while b
                 always (timestamp>= (funcall created-at-fn a) (funcall created-at-fn b)))
           (= (apply count-fn filters) (length want))
           (equal (mapcar (lambda (row) (id-string (funcall id-fn row))) paged)
                  (subseq full-ids start end))))))
